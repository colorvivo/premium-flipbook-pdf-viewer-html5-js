Code Example

const flipbook = new PremiumFlipBook({
  container: '#flipbook-container',
  pdfUrl: './document.pdf',

  ui: {
    navigation: {
      keyboard: true,        // Enable keyboard navigation
      arrows: true,          // Show arrow buttons
      clickToFlip: true,     // Click edges to flip
      mouseWheel: false      // Scroll to navigate
    }
  },

  // Accessibility options
  accessibility: {
    enabled: true,
    ariaLabels: true,       // Add ARIA labels
    focusIndicator: true,  // Visual focus ring
    announcePageChange: true // Screen reader announcements
  }
});
}
