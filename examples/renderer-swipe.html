Swipe Renderer

Touch-optimized horizontal swipe navigation for mobile

← Swipe left/right or drag with mouse →

Code Example

import PremiumFlipBook from 'premium-flipbook-pdf-viewer';

const flipbook = new PremiumFlipBook({
  container: '#flipbook-container',
  pdfUrl: './document.pdf',

  // Force Swipe renderer
  renderMode: 'swipe',

  // Swipe-specific options
  swipe: {
    threshold: 50,           // Min swipe distance (px)
    velocity: 0.3,           // Min velocity for quick swipe
    resistance: 0.8,         // Edge resistance factor
    animationDuration: 300   // Snap animation (ms)
  },

  // Single page mode works best for swipe
  singlePageMode: 'always',

  onReady: (fb) => {
    console.log('Renderer:', fb.getRendererType()); // 'swipe'
  }
});
