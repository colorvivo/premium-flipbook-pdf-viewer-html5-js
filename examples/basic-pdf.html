Basic PDF Source

Load and display a PDF file with automatic page rendering

Code Example

import PremiumFlipBook from 'premium-flipbook-pdf-viewer';

// Initialize flipbook with PDF source
const flipbook = new PremiumFlipBook({
  // Container element or selector
  container: '#flipbook-container',

  // PDF file URL
  pdfUrl: './assets/sample.pdf',

  // Renderer mode: 'auto', 'webgl', 'css', 'swipe'
  renderMode: 'auto',

  // Starting page
  startPage: 1,

  // UI configuration
  ui: {
    toolbar: { enabled: true },
    thumbnails: { enabled: true }
  },

  // Callbacks
  onReady: (fb) => {
    console.log('FlipBook ready!');
    console.log('Total pages:', fb.getPageCount());
  },

  onPageChange: ({ page, totalPages }) => {
    console.log(`Page ${page} of ${totalPages}`);
  },

  onError: (error) => {
    console.error('Error:', error);
  }
});
