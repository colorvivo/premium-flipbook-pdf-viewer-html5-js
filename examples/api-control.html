Code Example

// Navigation API
await flipbook.goToPage(5);      // Go to specific page
await flipbook.nextPage();       // Next page
await flipbook.prevPage();       // Previous page
await flipbook.firstPage();      // First page
await flipbook.lastPage();       // Last page

// Zoom API
flipbook.setZoom(2);             // Set zoom to 200%
flipbook.zoomIn();               // Zoom in by step
flipbook.zoomOut();              // Zoom out by step
flipbook.resetZoom();            // Reset to 100%

// Display & Transition
flipbook.setDisplayMode('single');  // Single page view
flipbook.setDisplayMode('double');  // Double page spread
flipbook.setTransition('slide');    // Set transition effect

// State getters
const page = flipbook.getCurrentPage();
const total = flipbook.getPageCount();
const zoom = flipbook.getZoom();
const mode = flipbook.getDisplayMode();
const renderer = flipbook.getRendererType();
const isFlipping = flipbook.isFlipping();

// UI toggles
flipbook.toggleThumbnails();
flipbook.toggleFullscreen();
}
