CSS 3D Renderer

CSS transform-based rendering with broad browser support

Code Example

import PremiumFlipBook from 'premium-flipbook-pdf-viewer';

const flipbook = new PremiumFlipBook({
  container: '#flipbook-container',
  pdfUrl: './document.pdf',

  // Force CSS renderer
  renderMode: 'css',

  // CSS-specific options
  css: {
    perspective: 2000,         // 3D perspective depth
    flipDuration: 600,        // Animation duration (ms)
    easing: 'ease-in-out',    // CSS easing function
    shadow: true,             // Page shadows
    gradientOverlay: true     // 3D gradient effect
  },

  onReady: (fb) => {
    console.log('Renderer:', fb.getRendererType()); // 'css'
  }
});

// Check CSS 3D support
if (PremiumFlipBook.isCSS3DSupported()) {
  // Safe to use CSS 3D transforms
}
