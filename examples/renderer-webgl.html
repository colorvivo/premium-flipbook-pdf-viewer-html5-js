WebGL 3D Renderer

High-performance GPU-accelerated rendering with Three.js

Code Example

import PremiumFlipBook from 'premium-flipbook-pdf-viewer';

const flipbook = new PremiumFlipBook({
  container: '#flipbook-container',
  pdfUrl: './document.pdf',

  // Force WebGL renderer
  renderMode: 'webgl',

  // WebGL-specific options
  webgl: {
    antialias: true,          // Smooth edges
    alpha: true,              // Transparent background
    maxTextureSize: 2048,     // Max texture resolution
    flipDuration: 800,        // Animation duration (ms)
    bendIntensity: 0.3,       // Page bend amount
    powerPreference: 'high-performance',

    // Lighting configuration
    lighting: {
      enabled: true,
      ambient: 0.6,           // Ambient light intensity
      directional: 0.4,       // Directional light
      position: [0, 1, 1]      // Light position
    }
  },

  onReady: (fb) => {
    console.log('Renderer:', fb.getRendererType()); // 'webgl'
  }
});

// Check WebGL support before initialization
if (PremiumFlipBook.isWebGLSupported()) {
  // Safe to use WebGL
}
