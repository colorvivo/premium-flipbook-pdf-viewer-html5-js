Events

Listen to flipbook events in real-time

Event Log

Code Example

// Subscribe to events
flipbook.on('pageChange', ({ page, previousPage, totalPages }) => {
  console.log(`Page changed: ${previousPage} → ${page}`);
});

flipbook.on('flipStart', ({ direction, fromPage }) => {
  console.log(`Flip started: ${direction}`);
});

flipbook.on('flipEnd', ({ page }) => {
  console.log(`Flip ended on page ${page}`);
});

flipbook.on('zoomChange', ({ zoom, previousZoom }) => {
  console.log(`Zoom: ${zoom * 100}%`);
});

flipbook.on('fullscreenChange', ({ isFullscreen }) => {
  console.log(`Fullscreen: ${isFullscreen}`);
});

// Subscribe once (auto-removes after first call)
flipbook.once('ready', ({ flipbook }) => {
  console.log('FlipBook initialized!');
});

// Unsubscribe
const handler = (data) => { ... };
flipbook.on('pageChange', handler);
flipbook.off('pageChange', handler);  // Remove specific handler
flipbook.off('pageChange');          // Remove all handlers
}
