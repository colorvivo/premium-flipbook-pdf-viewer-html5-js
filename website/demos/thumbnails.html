Thumbnails Panel

Visual page navigation with lazy-loaded thumbnails

Code Example

const flipbook = new PremiumFlipBook({
  container: '#flipbook-container',
  pdfUrl: './document.pdf',

  ui: {
    // Thumbnails configuration
    thumbnails: {
      enabled: true,
      position: 'left',       // 'left', 'right', 'bottom'
      width: 200,             // Panel width in px
      lazyLoad: true,         // Load thumbnails on demand
      preloadCount: 5         // Preload nearby thumbnails
    }
  }
});

// Toggle thumbnails programmatically
flipbook.toggleThumbnails();        // Toggle visibility
flipbook.toggleThumbnails(true);   // Force show
flipbook.toggleThumbnails(false);  // Force hide

// Listen to thumbnail toggle events
flipbook.on('thumbnailsToggle', ({ isOpen }) => {
  console.log(`Thumbnails ${isOpen ? 'opened' : 'closed'}`);
});
